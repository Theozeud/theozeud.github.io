% -------------------------------
% Entry geometry: a fixed date column on the left, indented body on the
% right -- the classic academic-CV layout. Everything lines up on the
% same left edge, which is what makes it easy to scan.
% -------------------------------
\newlength{\cvdatew}\setlength{\cvdatew}{3.2cm}
\newlength{\cvgap}\setlength{\cvgap}{0.6cm}
% geometry only sets the text width at \begin{document}, so compute the
% body column there -- otherwise it is sized from the class default.
\newlength{\cvbodyw}
\AtBeginDocument{\setlength{\cvbodyw}{\dimexpr\textwidth-\cvdatew-\cvgap\relax}}

% \cventry{title}{date}{sub-line}{note-line} (sub/note may be empty)
\newcommand{\cventry}[4]{%
  \noindent\begin{minipage}[t]{\cvdatew}\leavevmode\raggedleft\small\color{cvgray}#2\end{minipage}%
  \hspace{\cvgap}%
  \begin{minipage}[t]{\cvbodyw}\raggedright
    \textbf{#1}%
    \ifstrempty{#3}{}{\\[2pt]{\itshape\color{cvsub}#3}}%
    \ifstrempty{#4}{}{\\[2pt]{\small\color{cvgray}#4}}%
  \end{minipage}\par\vspace{0.65em}%
}

% \cvskill{label}{content} -- same left column, used for the Skills block
\newcommand{\cvskill}[2]{%
  \noindent\begin{minipage}[t]{\cvdatew}\leavevmode\raggedleft\small\scshape\color{cvgray}#1\end{minipage}%
  \hspace{\cvgap}%
  \begin{minipage}[t]{\cvbodyw}\raggedright #2\end{minipage}\par\vspace{0.45em}%
}
