% \cventry{title}{date}{sub-line}{note-line} (sub/note may be empty)
\newcommand{\cventry}[4]{%
  \noindent\begin{minipage}[t]{\cvdatew}\raggedleft\small\color{cvgray}#2\end{minipage}%
  \hspace{\cvgap}%
  \begin{minipage}[t]{\cvbodyw}\raggedright
    \textbf{#1}%
    \ifstrempty{#3}{}{\\[3pt]{\itshape\color{black!70}#3}}%
    \ifstrempty{#4}{}{\\[3pt]{\small\color{cvgray}#4}}%
  \end{minipage}\par\vspace{0.85em}%
}

% \cvskill{label}{content} -- same left column, used for the Skills block
\newcommand{\cvskill}[2]{%
  \noindent\begin{minipage}[t]{\cvdatew}\raggedleft\small\scshape\color{cvgray}#1\end{minipage}%
  \hspace{\cvgap}%
  \begin{minipage}[t]{\cvbodyw}\raggedright #2\end{minipage}\par\vspace{0.45em}%
}
